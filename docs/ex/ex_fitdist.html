Example with normal

# Example data set 
x = c(42,37,50,28,33,46,39,55,31,49,31,33,36,44,40.5,27,52,44.8,38,43,30,47,35,51)
hist(x) #ugly hist
# Fit distribution to data using maximum likelihood 
fit1 <- fitdist(x,distr= "norm", method = "mle")
# Generats plots comparising empirical cdf with fitted cdf, and histogram with fitted pdf
plot(fit1)
# Extract goodness of fit as an AIC-value 
AIC(fit1)
[1] 171.9233
# Extract the parameters for the fitted distribution. You use these values to sample or calculate things from the distribution 
fit1$estimate
     mean        sd 
40.095833  8.000181 
# For example, if the distribution is normal one can calculate the 95th percentile like this
qnorm(0.95,mean=fit1$estimate[1],sd=fit1$estimate[2])
[1] 53.25496
